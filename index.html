Google Sheets Integration Setup

Automatically store every submitted divination in your personal Google Sheet

Step-by-Step Instructions:

  1. Create a new Google Sheet in your Google Drive.
  2. In the sheet, click Extensions ➔ Apps Script.
  3. Delete any code in the Apps Script editor and paste the snippet below:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Append headers if row 1 is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Male Name", "Male DOB", "Female Name", "Female DOB", "Start Date", "Match Score %", "Zodiac Pair"]);
    }
    
    sheet.appendRow([
      new Date(),
      data.maleName || '',
      data.maleDob || '',
      data.femaleName || '',
      data.femaleDob || '',
      data.startDate || '',
      data.score || '',
      data.zodiacPair || ''
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
